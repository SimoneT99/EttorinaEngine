template <typename ButtonType>
inline bool ThreadSafeButtonManager<ButtonType>::check_event_presence(ButtonType button, Enums::ButtonEvent event) const{
    auto range = this->button_event_bucket.equal_range(button);
    for (auto it = range.first; it != range.second; ++it) {
            if (it->second == event) {
                return true;
            }
        }
    return false;
}

template <typename ButtonType>
inline void ThreadSafeButtonManager<ButtonType>::fire_event(Enums::MouseButton button, Enums::ButtonEvent event) const{
    auto it = this->button_event_bucket.find({button, event}); // Cerca la coppia chiave
    if (it != this->button_event_bucket.end()) {
        it->second->execute_command();
    }
}


/**
 * Polling methods
 */

template <typename ButtonType>
bool ThreadSafeButtonManager<ButtonType>::button_pressed_down(ButtonType button) const {
    return this->check_event_presence(button, Enums::ButtonEvent::PRESSED_DOWN); 
};

template <typename ButtonType>
bool ThreadSafeButtonManager<ButtonType>::button_pressed(ButtonType button) const {
    return this->check_event_presence(button, Enums::ButtonEvent::PRESSED);
};

template <typename ButtonType>
bool ThreadSafeButtonManager<ButtonType>::button_released(ButtonType button) const {
    return this->check_event_presence(button, Enums::ButtonEvent::RELEASED);
};

/**
 * Binding methods
 */

template <typename ButtonType>
void ThreadSafeButtonManager<ButtonType>::bind_button_event(ButtonType button, Enums::ButtonEvent event, std::unique_ptr<AbstractInputCommand> action){
    this->buttons_events_bindings.insert({std::make_pair(button, event), std::move(action)});
};

template <typename ButtonType>
void ThreadSafeButtonManager<ButtonType>::unbind_mouse_event(ButtonType button, Enums::ButtonEvent event) {
    this->buttons_events_bindings.erase({button, event});
};

/**
 * State management methods
 */

template <typename ButtonType>
void ThreadSafeButtonManager<ButtonType>::press_button(ButtonType button) {
    Enums::ButtonState state = this->get_mouse_button_state(button);
    switch (state)
        {
        case Enums::ButtonState::PRESSED:
            this->button_event_bucket.emplace(button, Enums::ButtonEvent::PRESSED);
            this->fire_event(button, Enums::ButtonEvent::PRESSED);
            break;
        case Enums::ButtonState::UNPRESSED:
            this->button_event_bucket.emplace(button, Enums::ButtonEvent::PRESSED_DOWN);
            this->button_state.emplace(button, Enums::ButtonState::PRESSED);
            this->fire_event(button, Enums::ButtonEvent::PRESSED_DOWN);
            break;
        default:
            std::cerr << "Invalid state, something went wrong at " << __FILE__ << " " << __LINE__ << std::endl;
            break; 
    }
};

template <typename ButtonType>
void ThreadSafeButtonManager<ButtonType>::release_button(ButtonType button) {
        Enums::ButtonState state = this->get_mouse_button_state(button);
    switch (state)
        {
        case Enums::ButtonState::PRESSED:
            this->button_event_bucket.emplace(button, Enums::ButtonEvent::RELEASED);
            this->button_state.emplace(button, Enums::ButtonState::UNPRESSED);
            this->fire_event(button, Enums::ButtonEvent::RELEASED);
            break;
        //case Enums::ButtonState::UNPRESSED:  
        //    break;
        default:
            std::cerr << "Invalid state, something went wrong at " << __FILE__ << " " << __LINE__ << std::endl;
            break; 
    }
};
